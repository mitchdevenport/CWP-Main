# green_party, restored from CWP 1.5 (master's common/parties/cwp_green_party.txt).
green_party = {
	color = { 29 168 36 }
	name = {
		first_valid = {
			triggered_desc = {
				desc = party_cwp_green_FRA
				trigger = { c:FRA ?= THIS }
			}
			triggered_desc = {
				desc = party_cwp_green_GER
				trigger = { OR = { c:GER ?= THIS c:AOZ ?= THIS c:BOZ ?= THIS c:FOZ ?= THIS } }
			}
			triggered_desc = {
				desc = party_cwp_green_DDR
				trigger = { c:DDR ?= THIS }
			}
			triggered_desc = {
				desc = party_cwp_green
				trigger = { always = yes }
			}
		}
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/agrarians.dds"
	}
	unlocking_technologies = {
		catalytic_convertors
	}
	valid_for_country = {
		always = yes
	}
	available_for_interest_group = {
		owner = {
			country_has_voting_franchise = yes
			NOT = { has_law = law_type:law_pollution_ban }
		}
		law_stance = { law = law_type:law_pollution_ban value >= neutral }
		# CWP 1.5's capitalists, ruralists opposed this: silence is not support
		OR = {
			NOT = { OR = { is_interest_group_type = ig_industrialists is_interest_group_type = ig_rural_folk } }
			law_stance = { law = law_type:law_pollution_ban value > neutral }
		}
	}
	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}
		if = {
			limit = { leader ?= { has_ideology = ideology:ideology_cwp_green } }
			add = {
				value = 6
				desc = "REASON_HEADLINER_IG"
			}
		}
		# The warmer the world, the more groups turn green
		if = {
			limit = { global_temp_target_celsius_cached >= 15 global_temp_target_celsius_cached < 16 }
			add = {
				value = 1
				desc = "REASON_GLOBAL_TEMPERATURE"
			}
		}
		if = {
			limit = { global_temp_target_celsius_cached >= 16 global_temp_target_celsius_cached < 17 }
			add = {
				value = 2
				desc = "REASON_GLOBAL_TEMPERATURE"
			}
		}
		if = {
			limit = { global_temp_target_celsius_cached >= 17 global_temp_target_celsius_cached < 18 }
			add = {
				value = 4
				desc = "REASON_GLOBAL_TEMPERATURE"
			}
		}
		if = {
			limit = { global_temp_target_celsius_cached >= 19 }
			add = {
				value = 8
				desc = "REASON_GLOBAL_TEMPERATURE"
			}
		}
		owner = {
			if = {
				limit = {
					has_variable = green_party_morale
				}
				add = {
					value = var:green_party_morale
					desc = "REASON_PARTY_MORALE"
				}
			}
			if = {
				limit = {
					has_variable = green_party_popularity
					has_variable = green_party_clout
				}
				add = {
					value = var:green_party_popularity
					desc = "REASON_PARTY_POPULARITY"
				}
				add = {
					value = var:green_party_clout
					desc = "REASON_PARTY_CLOUT"
				}
			}
		}
		if = {
			limit = { scope:number = 1 }
			multiply = {
				value = 2
				desc = "from_one_other_member"
			}
		}
		else_if = {
			limit = { scope:number = 2 }
			multiply = {
				value = 1.3
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = { scope:number = 3 }
			multiply = {
				value = 0.8
				desc = "from_three_prior_members"
			}
		}
		else_if = {
			limit = { scope:number = 4 }
			multiply = {
				value = 0.5
				desc = "from_more_than_three_prior_members"
			}
		}
		else_if = {
			limit = { scope:number >= 5 }
			multiply = {
				value = 0
				desc = "from_more_than_three_prior_members"
			}
		}
		# Victoria 3 1.5, which CWP 1.5 was made for, formed parties as election campaigns began
		if = {
			limit = {
				party ?= { is_party_type = green_party }
				owner = { in_election_campaign = no }
			}
			add = {
				value = 100
				desc = "REASON_PARTY_BETWEEN_CAMPAIGNS"
			}
		}
	}
}
