# monarchist_party, restored from CWP 1.5 (master's common/parties/cwp_monarchist_party.txt).
monarchist_party = {
	color = { 79 27 135 }
	name = {
		first_valid = {
			triggered_desc = {
				desc = party_cwp_monarchist_FRA
				trigger = { c:FRA ?= THIS }
			}
			triggered_desc = {
				desc = party_cwp_monarchist_GER
				trigger = { OR = { c:GER ?= THIS c:AOZ ?= THIS c:BOZ ?= THIS c:FOZ ?= THIS } }
			}
			triggered_desc = {
				desc = party_cwp_monarchist_DDR
				trigger = { c:DDR ?= THIS }
			}
			triggered_desc = {
				desc = party_cwp_monarchist
				trigger = { always = yes }
			}
		}
	}
	icon = {
		default = "/gfx/interface/icons/political_parties/conservatism.dds"
	}
	unlocking_technologies = {
	}
	valid_for_country = {
		always = yes
	}
	available_for_interest_group = {
		owner = {
			country_has_voting_franchise = yes
			OR = { has_modifier = recently_abolished_monarchy has_law = law_type:law_monarchy }
		}
		law_stance = { law = law_type:law_monarchy value >= neutral }
		law_stance = { law = law_type:law_universal_suffrage value <= neutral }
	}
	# Root - IG considering joining
	# scope:number - the number of IGs that would join this party before the root IG
	join_weight = {
		value = {
			value = -5
			desc = "base_value"
		}
		if = {
			limit = { is_interest_group_type = ig_landowners }
			add = {
				value = 6
				desc = "REASON_HEADLINER_IG"
			}
		}
		owner = {
			if = {
				limit = {
					has_variable = monarchist_party_morale
					has_variable = monarchist_party_cohesion
				}
				add = {
					value = var:monarchist_party_morale
					desc = "REASON_PARTY_MORALE"
				}
				add = {
					value = var:monarchist_party_cohesion
					desc = "REASON_PARTY_COHESION"
				}
			}
			if = {
				limit = {
					has_variable = monarchist_party_popularity
					has_variable = monarchist_party_clout
				}
				add = {
					value = var:monarchist_party_popularity
					desc = "REASON_PARTY_POPULARITY"
				}
				add = {
					value = var:monarchist_party_clout
					desc = "REASON_PARTY_CLOUT"
				}
			}
		}
		if = {
			limit = { scope:number = 1 }
			multiply = {
				value = 2
				desc = "from_one_other_member"
			}
		}
		else_if = {
			limit = { scope:number = 2 }
			multiply = {
				value = 1.3
				desc = "from_two_prior_members"
			}
		}
		else_if = {
			limit = { scope:number = 3 }
			multiply = {
				value = 0.8
				desc = "from_three_prior_members"
			}
		}
		else_if = {
			limit = { scope:number = 4 }
			multiply = {
				value = 0.5
				desc = "from_more_than_three_prior_members"
			}
		}
		else_if = {
			limit = { scope:number >= 5 }
			multiply = {
				value = 0
				desc = "from_more_than_three_prior_members"
			}
		}
		# Victoria 3 1.5, which CWP 1.5 was made for, formed parties as election campaigns began
		if = {
			limit = {
				party ?= { is_party_type = monarchist_party }
				owner = { in_election_campaign = no }
			}
			add = {
				value = 100
				desc = "REASON_PARTY_BETWEEN_CAMPAIGNS"
			}
		}
	}
}
