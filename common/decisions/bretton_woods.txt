sign_bretton_woods_agreement = {
	is_shown = {
		has_global_variable = bretton_woods_agreement
		exists = c:USA
		nor = {
			has_journal_entry = je_bretton_woods_agreement
			c:USA ?= this
		}
	}
	possible = {
		this = {
			is_subject = no
			government_legitimacy >= 15
			is_at_war = no
		}
	}
	when_taken = {
		# same as the starting signatories (activates the Pegged Currency law the agreement requires)
		effect_bretton_woods_signatory = yes
	}
	ai_chance = {
		value = 10
		if = {
			limit = {
				c:USA = {
					relations:root <= relations_threshold:poor
				}
			}
			add = -20
		}
		if = {
			limit = {
				c:USA = {
					relations:root >= relations_threshold:cordial
				}
			}
			add = 20
		}
	}
}

leave_bretton_woods_agreement = {
	is_shown = {
		has_global_variable = bretton_woods_agreement
		has_journal_entry = je_bretton_woods_agreement
		exists = c:USA
		not = {
			c:USA ?= this
		}
	}
	possible = {
		this = {
			government_legitimacy >= 10
			is_at_war = no
		}
	}
	when_taken = {
		remove_variable = maintain_pegged_currency_je_val
		remove_variable = pegged_currency
		remove_variable = peg_target_set
		activate_law = law_type:law_independent_monetary_policy
	}
	ai_chance = {
		value = -10
		if = {
			limit = {
				game_date > 1970.1.1
			}
			add = 50
		}
		if = {
			limit = {
				c:USA = {
					relations:root <= relations_threshold:poor
				}
			}
			add = 20
		}
	}
}

end_bretton_woods_agreement = {
	is_shown = {
		exists = c:USA
		c:USA ?= this
		or = {
			has_global_variable = bretton_woods_agreement
			has_journal_entry = je_bretton_woods_agreement
		}
	}
	possible = {
		this = {
			government_legitimacy >= 25
			is_subject = no
			is_at_war = no
			has_technology_researched = floated_exchange_rates
		}
	}
	when_taken = {
		custom_tooltip = {
			text = "WILL_END_BRETTON_WOODS"
			remove_variable = maintain_pegged_currency_je_val
			remove_variable = pegged_currency
			remove_variable = peg_target_set
			remove_global_variable = bretton_woods_agreement
			activate_law = law_type:law_independent_monetary_policy
		}
	}
	ai_chance = {
		value = -50
		if = {
			limit = {
				game_date > 1970.1.1
			}
			add = 50
		}
		if = {
			limit = {
				scaled_debt > 0.25
			}
			add = 50
		}
		if = {
			limit = {
				has_healthy_economy = no
			}
			add = 50
		}
	}
}
