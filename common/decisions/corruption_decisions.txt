anti_corruption_drive = {
	is_shown = {
		has_variable = var_corruption
	}

	possible = {
		custom_tooltip = {
			text = anti_corruption_drive_tooltip
			corruption_value >= 10
		}
		NOT = {
			has_modifier = anti_corruption_drive_conducted
		}
	}

	when_taken = {
		add_modifier = {
			name = anti_corruption_drive_conducted
			months = 60
		}
	}

	ai_chance = {
		value = 0
		# AI cheats
	}
}

anti_corruption_purge = {
	is_shown = {
		has_variable = var_corruption
		OR = {
			has_law = law_type:law_arbitrary_legalism
			has_law = law_type:law_puppet_judiciary
			has_law = law_type:law_secret_police
		}
	}

	possible = {
		custom_tooltip = {
			text = anti_corruption_purge_tooltip
			corruption_value >= 25
		}
		NOT = {
			has_modifier = anti_corruption_purge_conducted
		}
		OR = {
			has_law = law_type:law_arbitrary_legalism
			has_law = law_type:law_puppet_judiciary
			has_law = law_type:law_secret_police
		}
	}

	when_taken = {
		trigger_event = {
			id = corruption_events.1
		}
		add_modifier = {
			name = anti_corruption_purge_conducted
			months = 240
		}
	}

	ai_chance = {
		value = 0
		# AI cheats
	}
}
