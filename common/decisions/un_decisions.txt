ask_for_un_intervention = {
	is_shown = {
		is_at_war = yes
		has_variable = un_member
		NOT = {
			has_variable = asked_for_un_intervention
		}
	}

	possible = {
		custom_tooltip = { # the generated any_diplomatic_play tooltip has no perspective in a decision
			text = un_intervention_attacked_tt
			any_diplomatic_play = {
				target_is = root
				is_war = yes
			}
		}
		custom_tooltip = {
			text = is_un_member
			has_variable = un_member
		}
		NOT = {
			custom_tooltip = {
				text = hasnotaskedforintervention
				has_variable = asking_un_intervention
			}
		}
		NOR = {
			custom_tooltip = {
				text = doesnothavependingresolution
				has_global_variable = pending_resolution
			}
			custom_tooltip = {
				text = un_interventionvoteunderway
				has_global_variable = un_intervention_vote
			}
		}
	}

	when_taken = {
		effect_clear_un_intervention = yes
		set_variable = asking_un_intervention
		set_global_variable = {
			name = asking_un_intervention_country
			value = root
		}
		set_variable = {
			name = asked_for_un_intervention
			days = 200
		}
		every_country = {
			limit = {
				has_war_with = root
			}
			set_variable = un_intervention_enemy_country
		}
		set_global_variable = {
			name = pending_resolution_progress
			value = 0
		}
		set_global_variable = un_intervention_vote
		set_global_variable = pending_resolution
		custom_tooltip = {
			text = willsendunvote
			every_country = {
				limit = {
					has_variable = un_member
					NOT = { has_journal_entry = je_un_intervention }
				}
				add_journal_entry = {
					type = je_un_intervention
				}
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				any_scope_war = {
					has_war_support = {
						target = root
						value < 15
					}
				}
			}
			add = 50
		}
		if = {
			limit = {
				exists = c:STK
				exists = c:NRK
				OR = {
					AND = {
						c:STK ?= root
						has_war_with = c:NRK
					}
					AND = {
						c:NRK ?= root
						has_war_with = c:STK
					}
				}
			}
			add = 1000
		}
	}
}

join_un_ai = {
	is_shown = {
		is_ai = yes
		NOR = {
			has_variable = un_member
			has_variable = application_progress_un
		}
	}

	possible = {
		is_ai = yes
		is_at_war = no
		is_subject = no
		NOR = {
			has_variable = un_member
			has_variable = application_progress_un
			has_variable = un_application_rejected
		}
	}

	when_taken = {
		set_variable = {
			name = application_progress_un
			value = 0
		}
		set_variable = {
			name = un_supporting_list_size
			value = 0
		}
		add_to_global_variable_list = {
			name = un_application_list
			target = root
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				un_applying_states <= 3
			}
			add = 5
		}
		if = {
			limit = {
				un_applying_states <= 4
			}
			add = 5
		}
		if = {
			limit = {
				un_applying_states <= 5
			}
			add = 5
		}
		if = {
			limit = {
				un_applying_states > 5
			}
			add = -100
		}
	}
}
