adopt_independent_flag = {
	is_shown = {
		this = {
			is_subject = yes
			OR = {
				trigger_if = {
					limit = {
						exists = c:AST
					}
					c:AST ?= this
				}
				trigger_if = {
					limit = {
						exists = c:NZL
					}
					c:NZL ?= this
				}
				trigger_if = {
					limit = {
						exists = c:CAN
					}
					c:CAN ?= this
				}
				trigger_if = {
					limit = {
						exists = c:HND
					}
					c:HND ?= this
				}
			}
			NOT = { has_variable = independent_flag }
		}
	}

	possible = {
		is_subject_type = subject_type_dominion
	}

	when_taken = {
		this = {
			custom_tooltip = {
				text = adopt_independent_flag_tooltip
				set_variable = independent_flag
			}
		}
	}

	ai_chance = {
		value = 0
		if = {
			limit = {
				exists = c:HND
				c:HND ?= this
			}
			add = 100
		}
		if = {
			limit = {
				exists = c:CAN
				c:CAN ?= this
			}
			add = -10
		}
		if = {
			limit = {
				exists = c:AST
				c:AST ?= this
			}
			add = -80
		}
		if = {
			limit = {
				exists = c:NZL
				c:NZL ?= this
			}
			add = -70
		}
		if = {
			limit = {
				game_date > 1950.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 1960.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 1970.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 1980.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 1990.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 2000.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 2010.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 2020.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 2030.1.1
			}
			add = 10
		}
		if = {
			limit = {
				game_date > 2040.1.1
			}
			add = 10
		}
	}
}
