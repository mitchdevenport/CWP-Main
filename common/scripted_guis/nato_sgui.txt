#### Varcheck
nato_leader_varcheck = {
	scope = country
	is_shown = {
		has_variable = nato_leader
	}
}

nato_approve_membership_varcheck = {
	scope = country
	saved_scopes = {
		applicant_country_scope
	}
	is_shown = {
		is_in_nato = yes
		scope:applicant_country_scope = {
			NOT = {
				is_target_in_variable_list = {
					name = nato_supporting
					target = ROOT
				}
			}
		}
	}
}

##### Effects
nato_approve_membership = {
	scope = country
	saved_scopes = {
		applicant_country_scope
	}
	is_valid = {
		is_in_nato = yes
		scope:applicant_country_scope = {
			has_variable = application_progress_nato
		}
	}
	effect = {
		scope:applicant_country_scope = {
			if = {
				limit = {
					NOT = {
						is_target_in_variable_list = {
							name = nato_supporting
							target = ROOT
						}
					}
				}
				add_to_variable_list = {
					name = nato_supporting
					target = ROOT
				}
				change_variable = {
					name = nato_supporting_list_size
					add = 1
				}
			}
			# Withdraw an earlier rejection
			if = {
				limit = {
					is_target_in_variable_list = {
						name = nato_rejecting
						target = ROOT
					}
				}
				change_variable = {
					name = nato_supporting_list_size
					add = 1
				}
			}
			remove_list_variable = {
				name = nato_rejecting
				target = ROOT
			}
		}
	}
}

				target = ROOT
			}
		}
	}
}

####### Readiness
nato_low_readiness = {
	scope = country
	is_valid = {
		has_variable = nato_leader
		NOR = {
			has_global_variable = nato_low_mandated_readiness
			has_global_variable = nato_readiness_changed
		}
	}
	effect = {
		remove_global_variable = nato_medium_mandated_readiness
		remove_global_variable = nato_high_mandated_readiness
		set_global_variable = nato_low_mandated_readiness
		set_global_variable = {
			name = nato_readiness_changed
			days = 365
		}
	}
}
