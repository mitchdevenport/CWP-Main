raise_readiness = {
	scope = country

	is_valid = {
		readiness < 5
		NOT = {
			has_variable = changed_readiness
		}
	}

	effect = {
		change_variable = {
			name = var_readiness
			add = 1
		}
		set_variable = {
			name = changed_readiness
			days = 30
		}
		onaction_readiness_modifier_update = yes
	}
}

lower_readiness = {
	scope = country

	is_valid = {
		readiness > 1
		NOT = {
			has_variable = changed_readiness
		}
	}

	effect = {
		change_variable = {
			name = var_readiness
			subtract = 1
		}
		set_variable = {
			name = changed_readiness
			days = 30
		}
		onaction_readiness_modifier_update = yes
	}
}
