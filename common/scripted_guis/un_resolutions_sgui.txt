##### Effects
un_approve_membership = {
	scope = country
	saved_scopes = {
		applicant_country_scope
	}
	effect = {
		scope:applicant_country_scope = {
			add_to_variable_list = {
				name = un_supporting
				target = ROOT
			}
			if = { # the counter is only created with the application
				limit = { NOT = { has_variable = un_supporting_list_size } }
				set_variable = { name = un_supporting_list_size value = 0 }
			}
			change_variable = {
				name = un_supporting_list_size
				add = 1
			}
			if = {
				limit = {
					is_target_in_variable_list = {
						name = un_rejecting
						target = ROOT
					}
				}
				if = { # the counter is only created with the application
					limit = { NOT = { has_variable = un_supporting_list_size } }
					set_variable = { name = un_supporting_list_size value = 0 }
				}
				change_variable = {
					name = un_supporting_list_size
					add = 1
				}
			}
			remove_list_variable = {
				name = un_rejecting
				target = ROOT
			}
			remove_list_variable = {
				name = un_abstaining
				target = ROOT
			}
		}
	}
}

	scope = country
	effect = {
		add_modifier = declaration_of_human_rights
	}
}

add_opposing = {
	scope = country
	effect = {
		set_variable = is_opposing
		remove_variable = is_supporting
	}
}

add_supporting = {
	scope = country
	effect = {
		set_variable = is_supporting
		remove_variable = is_opposing
	}
}

add_abstaining = {
	scope = country
	effect = {
		remove_variable = is_supporting
		remove_variable = is_opposing
	}
}

add_veto = {
	scope = country
	is_valid = {
		has_variable = unsc_member
		has_global_variable = pending_resolution
	}
	effect = {
		set_global_variable = {
			name = vetoed
			value = 0
		}
	}
}
