effect_reset_un_lists = {
	set_global_variable = {
		name = pending_resolution_progress
		value = 0
	}
	every_country = {
		if = {
			limit = {
				has_variable = is_supporting
			}
			remove_variable = is_supporting
		}
		if = {
			limit = {
				has_variable = is_opposing
			}
			remove_variable = is_opposing
		}
	}
	remove_global_variable = asking_un_intervention_country
	remove_global_variable = vetoed
	remove_global_variable = pending_resolution
}

# Clears the bookkeeping of a finished UN intervention vote (requesting country and its enemies)
effect_clear_un_intervention = {
	every_country = {
		limit = {
			OR = {
				has_variable = asking_un_intervention
				has_variable = un_intervention_enemy_country
			}
		}
		remove_variable = asking_un_intervention
		remove_variable = un_intervention_enemy_country
	}
}

# The country asking for a UN intervention gives up its own war goals so that the ultimatum is a ceasefire on the status quo
# scope:asking_un_intervention_country = country that asked for the intervention
# scope:un_intervention_enemy_country = its enemy
effect_un_remove_wargoal_of_type = {
	every_diplomatic_play = {
		limit = {
			is_war = yes
			any_scope_play_involved = {
				this = scope:asking_un_intervention_country
			}
			scope:asking_un_intervention_country = {
				play_participant_has_war_goal_of_type_against = {
					type = $TYPE$
					target = scope:un_intervention_enemy_country
				}
			}
		}
		remove_war_goal = {
			who = scope:asking_un_intervention_country
			type = $TYPE$
		}
	}
}

##### Rebuild UN list
rebuild_un_memberlist = {
	clear_global_variable_list = un_members_list
	every_country = {
		limit = {
			has_variable = un_member
		}
		add_to_global_variable_list = {
			name = un_members_list
			target = THIS
		}
	}
}

##### Rebuild every list shown in the International Organizations and UN panels
rebuild_un_lists = {
	clear_global_variable_list = unsc_members_list
	clear_global_variable_list = un_members_list
	clear_global_variable_list = un_observers_list
	clear_global_variable_list = un_nonmember_list
	clear_global_variable_list = un_headquarters_holder
	clear_global_variable_list = un_application_list
	set_global_variable = {
		name = var_un_memberlist
		value = 0
	}
	every_country = {
		if = {
			limit = {
				has_variable = application_progress_un
			}
			add_to_global_variable_list = {
				name = un_application_list
				target = THIS
			}
		}
		if = {
			limit = {
				has_variable = unsc_member
			}
			add_to_global_variable_list = {
				name = unsc_members_list
				target = THIS
			}
		}
		if = {
			limit = {
				has_variable = un_member
			}
			add_to_global_variable_list = {
				name = un_members_list
				target = THIS
			}
			change_global_variable = {
				name = var_un_memberlist
				add = 1
			}
		}
		else_if = {
			limit = {
				has_variable = un_observer
			}
			add_to_global_variable_list = {
				name = un_observers_list
				target = THIS
			}
		}
		else = {
			add_to_global_variable_list = {
				name = un_nonmember_list
				target = THIS
			}
		}
		if = {
			limit = {
				has_variable = un_headquarters_holder
			}
			add_to_global_variable_list = {
				name = un_headquarters_holder
				target = THIS
			}
		}
	}
}
