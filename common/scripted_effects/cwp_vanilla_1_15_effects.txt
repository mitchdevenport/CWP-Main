# from 00_scripted_effects.txt
retired_commander_becomes_politician_roll_effect = {
	if = {
		limit = {
			NOR = {
				has_role_of_type = ruler
				has_role_of_type = heir
				has_role_of_type = politician
				has_role_of_type = agitator
				has_role_of_type = executive
				has_role_of_type = magnate
				is_interest_group_pool_full = character_role_politician
			}
			popularity >= define:NCharacters|RETIRED_COMMANDER_POLITICIAN_MIN_POPULARITY
		}
		random = {
			chance = {
				value = popularity
				multiply = define:NCharacters|RETIRED_COMMANDER_POLITICIAN_POPULARITY_CHANCE_FACTOR
				multiply = $CHANCE_MULT$
			}
			add_character_role = politician
			owner ?= {
				set_variable = {
					name = retired_commander_politician_cooldown
					years = retired_commander_politician_cooldown_years
				}
			}
		}
	}
}
