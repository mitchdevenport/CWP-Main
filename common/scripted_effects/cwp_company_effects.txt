# Victoria 3 1.14 gives every company a building of its own type (building_<company type>, building group
# bg_company_headquarter) in its home state. A civil war hands the government's states to the rebels together with
# these buildings; when the rebels win (or a secession succeeds) the old government's companies die but their buildings
# stay, and the state's new owner can never establish that company type there (error.log: "Tried to create building
# of Type building_company_... but such a building already exists"). Annexations do not need this: the game removes
# the buildings itself. Never remove a living company's building - the game then disbands the company.
# Add new company types to the list in cwp_remove_orphaned_company_buildings.

# Root = state, TYPE = company type
cwp_remove_orphaned_company_building = {
	if = {
		limit = {
			any_scope_building = { is_building_type = building_$TYPE$ }
			owner = {
				NOT = { has_company = company_type:$TYPE$ }
			}
		}
		remove_building = building_$TYPE$
	}
}

# Root = country: removes the company buildings in its states whose company type it does not have
cwp_remove_orphaned_company_buildings_in_country = {
	every_scope_state = {
		limit = {
			any_scope_building = { is_building_group = bg_company_headquarter }
		}
		cwp_remove_orphaned_company_buildings = yes
	}
}
