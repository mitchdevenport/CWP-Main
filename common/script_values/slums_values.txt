# Slums (informal settlements) - state scope. As in CWP 1.5, the game places building_slums levels up to this many
# (one state_slum_modifier a level, common/static_modifiers/cwp_restored_society_helpers.txt): more as the state's
# unemployment or the price of housing in its market rises. CWP 1.5 added every bracket that matched and divided the
# running total; one bracket applies here, the highest that matches.
state_slum_determiner = {
	value = state_population
	divide = 100000
	round = yes
}

state_max_slums = {
	value = 0
	if = {
		limit = {
			OR = {
				state_unemployment_rate >= 0.6
				owner.market ?= { mg:housing = { market_goods_pricier >= 0.7 } }
			}
		}
		add = state_slum_determiner
	}
	else_if = {
		limit = {
			OR = {
				state_unemployment_rate >= 0.4
				owner.market ?= { mg:housing = { market_goods_pricier >= 0.5 } }
			}
		}
		add = state_slum_determiner
		divide = 1.5
	}
	else_if = {
		limit = {
			OR = {
				state_unemployment_rate >= 0.3
				owner.market ?= { mg:housing = { market_goods_pricier >= 0.4 } }
			}
		}
		add = state_slum_determiner
		divide = 2
	}
	else_if = {
		limit = {
			OR = {
				state_unemployment_rate >= 0.2
				owner.market ?= { mg:housing = { market_goods_pricier >= 0.3 } }
			}
		}
		add = state_slum_determiner
		divide = 2.5
	}
	else_if = {
		limit = {
			OR = {
				state_unemployment_rate >= 0.1
				owner.market ?= { mg:housing = { market_goods_pricier >= 0.2 } }
			}
		}
		add = state_slum_determiner
		divide = 3
	}
	round = yes
}
