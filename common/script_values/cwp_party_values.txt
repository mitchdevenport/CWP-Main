# CWP 1.5's party values (master's common/script_values/politics_values.txt)

max_party_morale = 10
min_party_morale = -10

# Country scope: the popularity of the leaders of the party's interest groups
agrarian_party_popularity = {
	value = 0
	every_potential_party = {
		limit = { is_party_type = agrarian_party }
		every_member = {
			leader ?= {
				add = popularity
			}
		}
	}
	multiply = 0.1
}

# Country scope: +5 for every powerful interest group in the party, -1 for every other
agrarian_party_clout = {
	value = 0
	every_potential_party = {
		limit = { is_party_type = agrarian_party }
		every_member = {
			if = {
				limit = { is_powerful = yes }
				add = 5
			}
			else = {
				add = -1
			}
		}
	}
}
