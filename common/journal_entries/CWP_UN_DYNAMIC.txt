je_un_intervention = {
	icon = "gfx/interface/icons/cwp_icons/united_nations.dds"
	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
	possible = {
		OR = {
			has_variable = un_member
			has_variable = unsc_member
		}
		has_global_variable = un_intervention_vote
		NOT = {
			has_global_variable = vetoed
		}
	}
	immediate = {
		every_country = {
			limit = {
				has_variable = asking_un_intervention
			}
			save_scope_as = asking_un_intervention_country
		}
		if = {
			limit = {
				exists = scope:asking_un_intervention_country
			}
			every_country = {
				limit = {
					has_war_with = scope:asking_un_intervention_country
				}
				save_scope_as = un_intervention_enemy_country
			}
		}
	}
	current_value = {
		value = pending_resolution_progress_value
	}
	goal_add_value = {
		add = 100
	}
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = unsc_member
				}
				change_global_variable = {
					name = pending_resolution_progress
					add = 1
				}
			}
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				scope:asking_un_intervention_country ?= {
					is_at_war = no
				}
				AND = {
					scope:journal_entry = {
						is_goal_complete = yes
					}
					custom_tooltip = {
						text = "WILL_COMPLETE_UN"
						supporting_value_total > opposing_value_total
					}
				}
			}
		}
		effect = {
			hidden_effect = {
				if = {
					limit = {
						scope:asking_un_intervention_country ?= {
							is_at_war = yes
						}
					}
					every_country = {
						limit = {
							has_variable = un_member
						}
						trigger_event = {
							# Generic SUCCESS Event for all UN Members
							id = un_dynamic_resolution_events.1
						}
					}
				}
				else = {
					effect_clear_un_intervention = yes
				}
				remove_global_variable = un_intervention_vote
			}
			effect_reset_un_lists = yes
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				AND = {
					scope:journal_entry = {
						is_goal_complete = yes
					}
					custom_tooltip = {
						text = "WILL_FAIL_UN"
						opposing_value_total >= supporting_value_total
					}
				}
				custom_tooltip = {
					text = "IS_VETOED"
					has_global_variable = vetoed
				}
			}
		}
		effect = {
			hidden_effect = {
				every_country = {
					trigger_event = {
						# Generic FAIL Event for all countries
						id = un_dynamic_resolution_events.3
					}
				}
				remove_global_variable = un_intervention_vote
			}
			effect_reset_un_lists = yes
		}
	}

	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = {
				has_global_variable = un_intervention_vote
			}
		}
	}
	progressbar = yes
	# No can_deactivate: in 1.14 a deactivated journal entry added by effect never reactivates and blocks the next vote; 'invalid' removes it when the vote ends
	weight = 100
}
