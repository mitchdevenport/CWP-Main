je_interstate_highway_system = {
	icon = "gfx/interface/icons/building_icons/automobile_industry.dds"

	group = je_group_technology

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		has_technology_researched = automotive_cities
		NOT = { has_variable = cwp_interstate_highway_system_eligible }
	}

	possible = {
		has_technology_researched = automotive_cities
		custom_tooltip = {
			text = cwp_interstate_highway_eligibility_tt
			hidden_trigger = {
				any_scope_state = {
					filter = {
						is_incorporated = yes
					}
					percent >= 0.5
					any_scope_building = {
						is_building_type = building_urban_center
						level >= 5
					}
				}
				any_scope_state = {
					is_incorporated = yes
					state_population >= 1000000
					any_scope_building = {
						is_building_type = building_urban_center
						level >= 10
					}
				}
			}
		}
	}

	immediate = {
		set_variable = cwp_interstate_highway_system_eligible
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = cwp_interstate_highway_completion_tt
				hidden_trigger = {
					any_scope_state = {
						filter = {
							is_incorporated = yes
							state_population >= 1000000
						}
						percent >= 0.5
						has_building = building_interstate_highway_system
					}
					any_scope_state = {
						is_incorporated = yes
						state_population >= 1000000
						has_building = building_interstate_highway_system
					}
				}
			}
		}
		effect = {
			trigger_event = { id = cwp_interstate_highway_events.1 }
		}
	}


	weight = 10
	transferable = no
}
