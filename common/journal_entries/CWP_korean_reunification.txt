je_korean_reunification = {
	icon = "gfx/interface/icons/lens_toolbar_icons/nuclear_weapons.dds"
	should_be_pinned_by_default_uninvolved_or_context = yes
	
	group = je_group_historical_content

	activation_description = {
		always = yes
	}

	scripted_button = korean_reunification_button
	is_shown_when_inactive = {
		exists = c:NRK
		exists = c:STK
		OR = {
			#Is Korea
			c:NRK ?= this
			c:STK ?= this
			#Is Korea's overlord
			c:NRK = {
				is_subject_of = PREV
			}
			c:STK = {
				is_subject_of = PREV
			}
			#Is allied with Korea
			any_scope_ally = {
				c:NRK ?= this
			}
			any_scope_ally = {
				c:STK ?= this
			}
			#Is the superpower Korea is aligned with (CWP 1.5's alignment pact; in 1.14 the leader of Korea's power bloc)
			AND = {
				is_power_bloc_leader = yes
				is_in_same_power_bloc = c:NRK
			}
			AND = {
				is_power_bloc_leader = yes
				is_in_same_power_bloc = c:STK
			}
			#If an additional country is to be added per event | unused
			is_target_in_global_variable_list = {
				name = je_korean_reunification_can_vote
				target = THIS
			}
		}
	}
	possible = {
		custom_tooltip = {
			text = je_korean_reunification_reason_voting
		}
		exists = c:NRK
		exists = c:STK
		#Does not trigger if Korea is at war
		custom_tooltip = {
			text = cwp_koreas_not_diplomatic_play_targets_tt
			NOT = {
				c:NRK = {
					is_diplomatic_play_target = yes
				}
			}
			NOT = {
				c:STK = {
					is_diplomatic_play_target = yes
				}
			}
		}
		OR = {
			NOT = {
				custom_tooltip = {
					text = un_korean_reunification_not_failed_tt
					has_global_variable = korean_reunification_failed
				}
			}
			AND = {
				#Both Koreas have the same overlord
				c:STK = {
					overlord = {
						c:NRK = {
							is_subject_of = PREV
						}
					}
				}
				c:NRK = {
					overlord = {
						c:STK = {
							is_subject_of = PREV
						}
					}
				}
			}
		}
		
		OR = {
			AND = {
				#Both Koreas have the same overlord
				c:STK = {
					overlord = {
						c:NRK = {
							is_subject_of = PREV
						}
					}
				}
				c:NRK = {
					overlord = {
						c:STK = {
							is_subject_of = PREV
						}
					}
				}
			}
			AND = {
				#Both Koreas are independent and are both aligned with the same nation
				c:NRK = {
					is_subject = no
				}
				c:STK = {
					is_subject = no
				}
				c:NRK = {
					is_in_same_power_bloc = c:STK
				}
			}
			# Korean UN resolution Passed
			custom_tooltip = {
				text = un_korean_resolution_completed_tt
				has_global_variable = korean_resolution_complete
			}
		}
	}
	immediate = {
		add_to_global_variable_list = {
			name = je_korean_reunification_can_vote
			target = this
		}
		if = {
			limit = {
				c:STK ?= this
			}
			#Do this only once
			#Events simulate the AI voting behavior
			#For some reason only one event can be triggered per block
			hidden_effect = {
				every_in_global_list = {
					variable = je_korean_reunification_can_vote
					limit = {
						is_player = no
					}
					trigger_event = korean_reunification.101
				}
				every_in_global_list = {
					variable = je_korean_reunification_can_vote
					limit = {
						is_player = no
					}
					trigger_event = korean_reunification.102
				}
				every_in_global_list = {
					variable = je_korean_reunification_can_vote
					limit = {
						is_player = no
					}
					trigger_event = korean_reunification.103
				}
				every_in_global_list = {
					variable = je_korean_reunification_can_vote
					limit = {
						is_player = no
					}
					trigger_event = korean_reunification.104
				}
				every_in_global_list = {
					variable = je_korean_reunification_can_vote
					limit = {
						is_player = no
					}
					trigger_event = korean_reunification.105
				}
				every_in_global_list = {
					variable = je_korean_reunification_can_vote
					limit = {
						is_player = no
					}
					trigger_event = korean_reunification.106
				}
			}
		}
		if = {
			limit = {
				is_player = yes
			}
			trigger_event = korean_reunification.4
		}
	}
	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					c:STK ?= this
				}
				#Do this only once
				#Events simulate the AI voting behavior
				#For some reason only one event can be triggered per block
				hidden_effect = {
					every_in_global_list = {
						variable = je_korean_reunification_can_vote
						limit = {
							is_player = no
						}
						trigger_event = korean_reunification.101
					}
					every_in_global_list = {
						variable = je_korean_reunification_can_vote
						limit = {
							is_player = no
						}
						trigger_event = korean_reunification.102
					}
					every_in_global_list = {
						variable = je_korean_reunification_can_vote
						limit = {
							is_player = no
						}
						trigger_event = korean_reunification.103
					}
					every_in_global_list = {
						variable = je_korean_reunification_can_vote
						limit = {
							is_player = no
						}
						trigger_event = korean_reunification.104
					}
					every_in_global_list = {
						variable = je_korean_reunification_can_vote
						limit = {
							is_player = no
						}
						trigger_event = korean_reunification.105
					}
					every_in_global_list = {
						variable = je_korean_reunification_can_vote
						limit = {
							is_player = no
						}
						trigger_event = korean_reunification.106
					}
				}
			}
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			hidden_trigger = {
				always = no
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOT = {
					exists = c:STK
				}
				NOT = {
					exists = c:NRK
				}
			}
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			if = {
				limit = {
					c:STK ?= this
				}
				#Do this only once
				#Unifier
				if = {
					limit = {
						korean_reunification_total_vote_unifier_no_reunification > korean_reunification_total_vote_unifier_stk
						korean_reunification_total_vote_unifier_no_reunification > korean_reunification_total_vote_unifier_nrk
					}
					every_in_global_list = {
						variable = je_korean_reunification_can_vote
						trigger_event = korean_reunification.2
					}
					set_global_variable = korean_reunification_failed
				}
				else = {
					c:STK = {
						trigger_event = korean_reunification.1
					}
				}
			}
		}
	}
	timeout = 172
	weight = 100
}
