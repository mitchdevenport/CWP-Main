# Generic warfare journal entries, restored from the original mod where they had been commented out since its
# military rework. The original tracked barracks production methods per unit category (country-wide or on a front).
# They now follow the current military: barracks training production methods, combat unit types and mobilization
# options (see common/scripted_triggers/cwp_generic_warfare_triggers.txt). Communications and the equipment
# generations without a counterpart in the current tree were not restored.


############ Infantry ############

je_post_war_infantry = {
	icon = "gfx/interface/icons/event_icons/event_military.dds"

	group = je_group_technology

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		NOT = { has_variable = post_war_infantry_je_done }
	}

	possible = {
		has_technology_researched = divisional_organization
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			OR = {
				cwp_gw_barracks_training_adopted = { PM = pm_codified_instruction_barracks }
				AND = {
					NOT = { has_variable = post_war_infantry_je_done }
					has_technology_researched = mobile_warfare
				}
			}
		}
		effect = {
			trigger_event = { id = generic_military_events.1 }
			set_variable = post_war_infantry_je_done
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			set_variable = post_war_infantry_je_done
		}
	}


	timeout = 3650		# 10 years

}
