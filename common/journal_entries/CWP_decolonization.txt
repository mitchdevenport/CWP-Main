je_decolonization = {
	icon = "gfx/interface/icons/event_icons/event_industry.dds"

	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}

	is_shown_when_inactive = {
		NOR = {
			country_rank = rank_value:superpower
			has_modifier = sun_never_sets
			has_modifier = sun_setting_on_empire
		}
		any_country = {
			is_subject_of = root
			NOR = {
				has_variable = special_decolonization_scenario
				is_subject_type = subject_type_military_occupation
			}
		}
		NOT = {
			has_journal_entry = je_gaddafi_unification
		}
	}

	possible = {
		OR = {
			country_rank < rank_value:minor_power
			has_variable = cwp_decolonization_pending # 1946 colonial empires and costly colonial wars (effect_decolonization_after_war)
		}
	}

	immediate = {
		if = {
			limit = { has_variable = cwp_decolonization_pending }
			remove_variable = cwp_decolonization_pending
		}
		if = {
			limit = { has_variable = cwp_decolonization_tracking }
			remove_variable = cwp_decolonization_tracking
		}
		if = {
			limit = {
				NOT = {
					has_variable = imperial_unity
				}
			}
			set_variable = {
				name = imperial_unity
				value = 100
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = superpower_pressure
				}
			}
			set_variable = {
				name = superpower_pressure
				value = 0
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = domestic_pressure
				}
			}
			set_variable = {
				name = domestic_pressure
				value = 0
			}
		}
		if = {
			limit = {
				NOT = {
					has_variable = imperial_confidence
				}
			}
			set_variable = {
				name = imperial_confidence
				value = 0
			}
		}
	}

	scripted_button = decolonization_abandon_empire_button
	scripted_button = decolonization_imperial_conference_button

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_decolonization_complete_condition_tooltip
				# CWP 1.5 also completed it on reaching superpower rank, which the 1.14 United Kingdom does by 1948, ending its
				# twenty years (and its India factors to 1956) almost at once; the empire's fate is decided when time runs out
				always = no
			}
		}
		effect = {
			custom_tooltip = {
				text = je_decolonization_complete_tooltip
				if = {
					limit = {
						var:imperial_unity = 100
					}
					trigger_event = {
						id = decolonization.1 # Keep the Empire Together
						days = 0
					}
				}
				else_if = {
					limit = {
						var:imperial_unity > 50
					}
					trigger_event = {
						id = decolonization.2 # Amicable Separation
						days = 0
					}
				}
				else = {
					trigger_event = {
						id = decolonization.3 # Clean Break
						days = 0
					}
				}
				effect_remove_imperial_conference_modifiers = yes
			}
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			custom_tooltip = {
				text = je_decolonization_fail_tooltip
				var:imperial_unity <= 0
			}
		}
		effect = {
			trigger_event = {
				id = decolonization.101 # We have lost control
				days = 0
			}
			hidden_effect = {
				effect_remove_imperial_conference_modifiers = yes
			}
		}
	}

	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			NOT = {
				any_country = {
					has_subject_relation_with = root
					NOR = {
						has_variable = special_decolonization_scenario
						is_subject_type = subject_type_military_occupation
					}
				}
			}
		}
		effect = {
			effect_decolonize_all = yes
			effect_remove_imperial_conference_modifiers = yes
		}
	}

	outcome = {
		header = JOURNAL_ENTRY_TIMEOUT
		effect_header = JOURNAL_ENTRY_ON_TIMEOUT
		conclusion_desc = JOURNAL_ENTRY_TIMEOUT_CONCLUSION_DESC
		flags = { timeout }
		effect = {
			custom_tooltip = {
				text = je_decolonization_timeout_tooltip
				if = {
					limit = {
						var:imperial_unity = 100
					}
					trigger_event = {
						id = decolonization.1 # Keep the Empire Together
						days = 0
					}
				}
				else_if = {
					limit = {
						var:imperial_unity > 50
					}
					trigger_event = {
						id = decolonization.2 # Amicable Separation
						days = 0
					}
				}
				else = {
					trigger_event = {
						id = decolonization.3 # Clean Break
						days = 0
					}
				}
				effect_remove_imperial_conference_modifiers = yes
			}
		}
	}






	status_desc = DECOLONIZATION_PROGRESS_DESC

	timeout = 7300


	on_monthly_pulse = {
		effect = {
			set_variable = cwp_decolonization_tracking
			change_variable = {
				name = imperial_unity
				add = imperial_unity_growth
			}
			clamp_variable = {
				name = imperial_unity
				min = 0
				max = 100
			}
			change_variable = {
				name = superpower_pressure
				add = superpower_pressure_growth
			}
			clamp_variable = {
				name = superpower_pressure
				min = -0.5
				max = 0.5
			}
			change_variable = {
				name = domestic_pressure
				add = domestic_pressure_growth
			}
			clamp_variable = {
				name = domestic_pressure
				min = 0
				max = 1
			}
		}
	}

	current_value = {
		value = 0
		# CWP 1.5 started the entry from history with a goal baseline of 0, so the bar shows unity out of 100. 1.14 records
		# the baseline after immediate (unity 100), so unity only counts from the entry's first monthly update
		if = {
			limit = {
				has_variable = imperial_unity
				has_journal_entry = je_decolonization
				has_variable = cwp_decolonization_tracking
			}
			value = var:imperial_unity
		}
	}

	goal_add_value = {
		value = 100
	}

	weight = 1000
	transferable = no
	can_revolution_inherit = yes
	progressbar = yes
	can_deactivate = no
	should_be_pinned_by_default_uninvolved_or_context = yes
}
