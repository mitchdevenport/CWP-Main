# triggered by UN Resolution Screen and the nuclear program decisions
je_acquire_civilian_nuclear_capability = {
	icon = "gfx/interface/icons/lens_toolbar_icons/nuclear_weapons.dds"

	group = je_group_technology

	activation_description = {
		always = yes
	}

	should_be_pinned_by_default_uninvolved_or_context = yes

	scripted_button = pause_nuclear_program_button
	scripted_button = resume_nuclear_program_button
	scripted_button = conduct_nuclear_test_button

	possible = {
		custom_tooltip = {
			text = "nuclear_in_progress"
			has_modifier = developing_civilian_nuclear_capability
		}
	}

	immediate = {
		nuclear_program_je_initialize = yes
	}

	current_value = {
		value = nuclear_program_progress_sv
	}

	goal_add_value = {
		value = nuclear_program_goal_add_sv
	}

	on_monthly_pulse = {
		effect = {
			if = {
				limit = {
					var:nuclear_program_progress < 80
				}
				change_variable = {
					name = nuclear_program_progress
					add = nuclear_progress_increase_per_month
				}
			}
		}
	}

	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
		}
		effect = {
			add_modifier = {
				name = civilian_nuclear_capable
			}
			remove_modifier = developing_civilian_nuclear_capability
			remove_variable = nuclear_program_progress
			remove_variable = nuclear_program_paused
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				NOR = {
					has_law = law_type:law_civilian_applications
					has_law = law_type:law_deterrence
					has_law = law_type:law_first_strike
				}
				canceled_nuclear_program = yes
			}
		}
		effect = {
			remove_variable = nuclear_test_pending
			remove_modifier = developing_civilian_nuclear_capability
			remove_variable = nuclear_program_progress
			remove_variable = nuclear_program_paused
		}
	}




	progressbar = yes

	weight = 100
}
