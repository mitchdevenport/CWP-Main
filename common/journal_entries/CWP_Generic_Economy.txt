# Housing For The People
# CWP 1.5: well-housed states, few slums and cheap housing
je_housing = {
	icon = "gfx/interface/icons/event_icons/event_trade.dds"
	
	group = je_group_internal_affairs
	
	is_shown_when_inactive = {
		is_player = yes
		NOR = {
			has_variable = housing_je_done
			pop_type_percent_country = {
				pop_type = peasants
				percent > 0.6
			}
		}
		any_scope_building = {
			is_building_type = building_urban_center
			level >= 5
		}
	}
	immediate = {
		if = {
			limit = {
				market_capital.market ?= {
					mg:housing = {
						market_goods_cheaper < 0
					}
				}
			}
			set_variable = market_goods_more_expensive
		}
		else_if = {
			limit = {
				market_capital.market ?= {
					mg:housing = {
						market_goods_cheaper > 0
					}
				}
			}
			set_variable = market_goods_cheaper
		}
	}
	possible = {
		any_scope_building = {
			is_building_type = building_housing
		}
	}
	complete = {
		any_scope_state = {
			percent >= 0.95
			has_building = building_housing
			any_scope_building = {
				is_building_type = building_housing
				occupancy < 0.95
				level >= 10
			}
		}
		any_scope_state = {
			percent >= 0.9
			OR = {
				NOT = { has_building = building_slums }
				any_scope_building = {
					is_building_type = building_slums
					occupancy <= 0.1
				}
			}
		}
		market_capital.market ?= {
			mg:housing = {
				market_goods_cheaper > 0
			}
		}
	}
	on_complete = {
		trigger_event = {
			id = cwp_generic_economy_events.1
		}
		set_variable = housing_je_done
	}
}

	group = je_group_internal_affairs

	is_shown_when_inactive = {
		is_player = yes
		NOT = {
			has_variable = high_speed_rail_je_done
		}
	}
	possible = {
		has_technology_researched = high_speed_rail
		any_scope_building = {
			is_building_type = building_railway
			has_active_production_method = pm_high_speed_rail
		}
	}
	complete = {
		any_scope_state = {
			percent >= 0.75

				is_building_type = building_railway
				has_active_production_method = pm_high_speed_rail
				occupancy >= 0.75
				level >= 5
			}
		}
		trigger_if = {
			limit = {
				any_scope_building = {
					OR = {
						is_building_type = building_coffee_plantation

						is_building_type = building_tea_plantation
						is_building_type = building_tobacco_plantation
						is_building_type = building_sugar_plantation
						is_building_type = building_banana_plantation
						is_building_type = building_coal_mine
						is_building_type = building_oil_rig
						is_building_type = building_iron_mine
						is_building_type = building_bauxite_mine
						is_building_type = building_uranium_mine
						is_building_type = building_lead_mine
						is_building_type = building_sulfur_mine
						is_building_type = building_gold_mine
						is_building_type = building_logging_camp
					}
				}
			}
			any_scope_building = {
				filter = {
					OR = {
						is_building_type = building_coffee_plantation
						is_building_type = building_cotton_plantation

						is_building_type = building_uranium_mine
						is_building_type = building_lead_mine
						is_building_type = building_sulfur_mine
						is_building_type = building_gold_mine
						is_building_type = building_logging_camp
					}
				}
				percent > 0.5
				OR = {
					has_active_production_method = pm_intermodal_containers_plantation
					has_active_production_method = pm_cold_chain_logistics_plantation
					has_active_production_method = pm_intermodal_containers_mines
