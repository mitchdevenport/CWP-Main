# triggered by UN Resolution Screen
je_declaration_of_human_rights = {
	icon = "gfx/interface/icons/cwp_icons/united_nations.dds"
	group = je_group_foreign_affairs

	activation_description = {
		always = yes
	}
	should_be_pinned_by_default_uninvolved_or_context = yes
	possible = {
		OR = {
			has_variable = un_member
			has_variable = unsc_member
		}
		NOR = {
			has_global_variable = declaration_of_human_rightscomplete
			has_global_variable = declaration_of_human_rightsfail
			has_global_variable = vetoed
		}
	}
	current_value = {
		value = pending_resolution_progress_value
	}
	goal_add_value = {
		add = 100
	}
	on_weekly_pulse = {
		effect = {
			if = {
				limit = {
					has_variable = unsc_member
				}
				change_global_variable = {
					name = pending_resolution_progress
					add = 0.5
				}
			}
		}
	}
	immediate = {
		every_country = {
			limit = {
				has_variable = un_member
			}
			save_scope_as = s_un_memberstates
		}
	}
	outcome = {
		header = je_standard_completion_header
		effect_header = je_standard_on_completion_header
		color = "gold_positive"
		conclusion_desc = je_standard_completion_conclusion_desc
		trigger = {
			scope:journal_entry = {
				is_goal_complete = yes
			}
			custom_tooltip = {
				text = "WILL_COMPLETE_UN"
				supporting_value_total > opposing_value_total
			}
		}
		effect = {
			hidden_effect = {
				every_country = {
					trigger_event = {
						id = un_resolution_events.1
					}
				}
				every_country = {
					limit = {
						has_variable = un_member
					}
					add_modifier = {
						name = declaration_of_human_rights
					}
				}
			}
			effect_reset_un_lists = yes
			set_global_variable = declaration_of_human_rightscomplete
		}
	}

	outcome = {
		header = je_standard_failure_header
		effect_header = je_standard_on_failure_header
		color = "blue_negative"
		conclusion_desc = je_standard_failure_conclusion_desc
		trigger = {
			OR = {
				AND = {
					scope:journal_entry = {
						is_goal_complete = yes
					}
					custom_tooltip = {
						text = "WILL_FAIL_UN"
						opposing_value_total >= supporting_value_total
					}
				}
				custom_tooltip = {
					text = "IS_VETOED"
					has_global_variable = vetoed
				}
			}
		}
		effect = {
			hidden_effect = {
				every_country = {
					trigger_event = {
						id = un_resolution_events.2
					}
				}
			}
			effect_reset_un_lists = yes
			set_global_variable = {
				name = declaration_of_human_rightsfail
				days = 365
			}
		}
	}

	outcome = {
		conclusion_desc = je_standard_failure_conclusion_desc
		flags = { hidden invalid_outcome }
		trigger = {
			OR = {
				has_global_variable = declaration_of_human_rightscomplete
				has_global_variable = declaration_of_human_rightsfail
			}
		}
	}
	progressbar = yes
	# No can_deactivate: in 1.14 a deactivated journal entry added by effect never reactivates and blocks the next vote; 'invalid' removes it when the vote ends
	weight = 100
}
