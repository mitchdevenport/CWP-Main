# Nationalization, restored from CWP 1.5 (master's common/laws/cwp_11_nationalization.txt).
# CWP 1.5 enforced these laws through ownership production methods: each step down the ladder made the state run one
# more sector (utilities and infrastructure < resource extraction < military industry < heavy industry < everything).
# Victoria 3 1.14 has building ownership instead, so the ladder now works through its ownership rules:
#  - No Nationalization forbids nationalization and puts state-owned buildings up for privatization
#  - each further step makes nationalization cheaper and less radicalising; heavy industry and total nationalization
#    stop privatization outside companies
#  - the AI's nationalization desire follows the law (common/ai_strategies/00_default_strategy.txt)
# CWP 1.5's private construction allocation (0.75 for no nationalization down to 0 for total) is now set mainly by the
# economic system law, so these laws only shift it in the same order.
# CWP 1.5's AI moved one step up the ladder under State Corporations; the current analogue is the command economy.
# The ladder is a can_enact condition (unlocking_laws within the group left countries created during the game without
# a law), and No Nationalization comes first because the game gives new countries the first law of each group.

law_no_nationalization = {
	group = lawgroup_nationalization
	icon = "gfx/interface/icons/law_icons/laissez_faire.dds"
	progressiveness = 0

	# CWP 1.5's ladder: only a neighbouring step can be enacted (the engine also requires this of the
	# current law, hence the law itself; a new country without a law in the group may take any)
	can_enact = {
		OR = {
			has_law = law_type:law_no_nationalization
			has_law = law_type:law_public_utility_companies
			hidden_trigger = { cwp_has_no_nationalization_law = yes }
		}
	}

	modifier = {
		country_private_construction_allocation_mult = 0.1
		state_infrastructure_from_population_mult = 0.1
		country_disable_nationalization_bool = yes
		country_force_privatization_bool = yes
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

law_public_utility_companies = {
	group = lawgroup_nationalization
	icon = "gfx/interface/icons/law_icons/interventionism.dds"
	progressiveness = 0

	# CWP 1.5's ladder: only a neighbouring step can be enacted (the engine also requires this of the
	# current law, hence the law itself; a new country without a law in the group may take any)
	can_enact = {
		OR = {
			has_law = law_type:law_public_utility_companies
			has_law = law_type:law_resource_extraction_nationalization
			has_law = law_type:law_no_nationalization
			hidden_trigger = { cwp_has_no_nationalization_law = yes }
		}
	}

	modifier = {
		country_bureaucrats_pol_str_mult = 0.1
		country_gold_reserve_limit_mult = 0.1
	}

	ai_enact_weight_modifier = {
		value = 0
		if = {
			limit = {
				has_law_or_variant = law_type:law_command_economy
				NOR = {
					has_law = law_type:law_total_nationalization
					has_law = law_type:law_heavy_industry_nationalization
					has_law = law_type:law_military_industry_nationalization
					has_law = law_type:law_resource_extraction_nationalization
				}
			}
			add = 100
		}
		if = { #Petitions
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
