# Centralization, restored from CWP 1.5 (master's common/laws/cwp_04_centralization.txt).
# Victoria 3 1.14 adaptations:
#  - law enactment time t -> law enactment speed 1/(1+t) - 1
#  - decree cost is a state modifier
#  - radicals from discrimination -> radicalism increases of the four discriminated acceptance statuses
#  - CWP 1.5's pop support (farmers for local power, service workers for central power) now comes from
#    interest group ideologies (see the lawgroup_centralization stances)
#  - the ladder is a can_enact condition: as unlocking_laws within the group it left countries created during the
#    game without any law in it (the game gives new countries the first law of each group they can have)

law_confederated = {
	group = lawgroup_centralization
	icon = "gfx/interface/icons/law_icons/confederated.dds"
	progressiveness = 0

	# CWP 1.5's ladder: only a neighbouring step can be enacted (the engine also requires this of the
	# current law, hence the law itself; a new country without a law in the group may take any)
	can_enact = {
		OR = {
			has_law = law_type:law_confederated
			has_law = law_type:law_federalized
			hidden_trigger = { cwp_has_no_centralization_law = yes }
		}
	}

	modifier = {
		country_authority_mult = -0.5
		state_decree_cost_mult = -0.9
		country_institution_size_change_speed_mult = -0.5
		country_law_enactment_speed_mult = -0.5
		country_secession_progress_mult = -0.5
		state_bureaucracy_population_base_cost_factor_mult = -0.25
		state_radicalism_increases_second_rate_citizen_mult = -0.3
		state_radicalism_increases_open_prejudice_mult = -0.3
		state_radicalism_increases_violent_hostility_mult = -0.3
		state_radicalism_increases_cultural_erasure_mult = -0.3
	}

	ai_will_do = {
		num_taxed_goods = 0 # Only if they have no consumption taxes
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
