# Recreational drugs, restored from CWP 1.5 (master's common/laws/cwp_27_recreational_drugs.txt).
# The current goods have no opium, so the laws ban and allow tobacco, liquor and wine; CWP 1.5's distilleries are the
# beverage industries. CWP 1.5's public support for legal tobacco and alcohol now comes from interest group ideologies.
# Alcohol Allowed comes first because the game gives new countries the first law of each group; the stricter laws
# follow, then All Drugs Allowed.

law_alcohol_allowed = {
	group = lawgroup_recreational_drugs
	icon = "gfx/interface/icons/law_icons/traditionalism.dds"
	progressiveness = 0

	on_activate = {
		effect_cwp_allow_recreational_good = { GOODS = tobacco }
		effect_cwp_allow_recreational_good = { GOODS = liquor }
		effect_cwp_allow_recreational_good = { GOODS = wine }
	}

	modifier = {
		state_mortality_mult = 0.1
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}

	}
}
