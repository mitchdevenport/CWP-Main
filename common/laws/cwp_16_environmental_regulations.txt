# Environmental regulations, restored from CWP 1.5 (master's common/laws/cwp_16_environmental_regulations.txt).
# As in CWP 1.5 the laws have no modifiers of their own: each sets the lowest green technology industries may use
# (pmg_green_tech_industry, common/production_methods/cwp_green_tech.txt), and a pollution ban moves combustion power
# to renewables (effect_ban_pollution_industries).
# Technologies mapped onto the current tree: CWP 1.5's early climatology -> clean air measures; its IPCC technology ->
# climate sciences for emission regulation. Net zero and the pollution ban require urban ecology, which also unlocks the
# advanced green technology they mandate (CWP 1.5 allowed them before that technology existed).

law_no_environment_regulations = {
	group = lawgroup_environmental_regulations
	icon = "gfx/interface/icons/law_icons/legacy_slavery.dds"
	progressiveness = -10

	# AI will never regress to no regulations (will destroy the game)
	ai_will_do = {
		always = no
	}

	ai_enact_weight_modifier = { #Petitions
		value = 0
		if = {
			limit = { ai_has_enact_weight_modifier_journal_entries = yes }
			add = 750
		}
	}
}
