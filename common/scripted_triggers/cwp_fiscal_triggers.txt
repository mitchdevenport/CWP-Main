is_leading_central_bank = {
	is_target_in_global_variable_list = {
		name = central_bank_list
		target = THIS
	}
}

# Vanilla's concept_economy_health: healthy if fixed incomes exceed fixed expenses or the principal is under half the credit.
# The built-in has_healthy_economy reads hidden AI/interface state in 1.14 (assertion, multiplayer desync), so gamestate uses this.
cwp_has_healthy_economy = {
	OR = {
		net_fixed_income > 0
		scaled_debt < 0.5
	}
}

# The United States while Bretton Woods holds: the agreement fails for every signatory once US inflation leaves -0.5 to 5%
# (je_bretton_woods_agreement), so the AI's monetary policy keeps it inside (ai_wants_contractionary/stable/expansionary)
cwp_ai_defends_bretton_woods = {
	has_global_variable = bretton_woods_agreement
	c:USA ?= THIS
}

currency_leader_exists = {
	any_country = {
		is_leading_central_bank = yes
		var:cwp_currency = flag:currency_$currency$
	}
}

can_adopt_currency = {
	or = {
		and = {
			not = {
				var:cwp_currency = flag:currency_dem
			}
			c:GER ?= THIS
		}
		and = {
			not = {
				var:cwp_currency = flag:currency_aud
			}
			year >= 1960
			c:AST ?= THIS
		}
		and = {
			not = {
				var:cwp_currency = flag:currency_nzd
			}
			year >= 1960
			c:NZL ?= THIS
		}
	}
}
