# from 00_ep2_victoria_scripted_triggers.txt
character_has_valid_reason_for_coup = {
    save_temporary_scope_as = coupist_to_be
    OR = {
        owner ?= { # Case 1 - owner is enacting a law the character does not approve of
            save_temporary_scope_as = couped_to_be
            exists = currently_enacting_law
            OR = {
                scope:coupist_to_be ?= {
                    law_enactment_stance = { # Refers to any_scope_character
                        law = scope:couped_to_be.currently_enacting_law.type # Refers to owner
                        value < neutral # Disapproves of the law
                    }
                }
                AND = {
                    scope:coupist_to_be ?= {
                        law_enactment_stance = { # Refers to any_scope_character
                            law = scope:couped_to_be.currently_enacting_law.type # Refers to owner
                            value < approve # Character does not approve of the law
                        }
                        interest_group ?= {
                            law_enactment_stance = { # Refers to any_scope_character's interest group
                                law = scope:couped_to_be.currently_enacting_law.type # Refers to owner
                                value < neutral # Interest group opposes the law
                            }
                        }
                    }
                }
            }
        }
        this.interest_group = {
            OR = {
                has_modifier = modifier_failed_to_meet_petition # Case 2 - owner has failed a petition for this.interest_group
                has_modifier = negotiation_failed_quest # Case 3 - owner has failed a negotiation for this.interest_group
            }
        }
        OR = {
            has_trait = special_character_napoleonic_return # Case 4 - this just has a coupiful personality
            has_trait = special_character_blanqui_is_free
        }
        # Not in this trigger, but exposed elsewhere:
        # Case 5 - Foreign-Backed Coup
        # Case 6 - Foreign-Backed Coup
    }
}

# from ip4_negotiation_triggers.txt
ig_could_coup_government = {
    owner = {
        NOT = { has_variable = coup_cooldown_var }
        NOT = { has_variable = coup_immunity }
    }
    any_scope_character = {
        OR = {
            has_role_of_type = general
            has_role_of_type = admiral
        }
        loyalty < define:NCharacters|LOYALTY_UNCERTAIN_THRESHOLD
        has_variable = monthly_cached_commander_coup_strength
        var:monthly_cached_commander_coup_strength >= {
            value = owner.country_coup_resistance
            multiply = 0.9
        }
        character_has_valid_reason_for_coup = yes
    }
}

# from 00_ai_triggers.txt
ai_ceding_state_would_isolate_states = {
	$STATE$ = { has_port_state = yes }
	$RECEIVER$ = {
		NOT = { market = $CEDER$.market }
	}
	NOT = {
		$CEDER$.market = {
			any_scope_country = {
				any_scope_state = {
					NOT = { this = $STATE$ }
					has_port_state = yes
					is_in_same_market_area = $STATE$
				}
			}
		}
	}
	OR = {
		AND = {
			$STATE$ = {
				NOT = { is_in_same_market_area = $CEDER$.capital }
				any_neighbouring_state = {
					is_in_same_market_area = $STATE$
				}
			}
		}
		AND = {
			$STATE$ = { is_in_same_market_area = $CEDER$.capital }
			$CEDER$ = {
				any_scope_state = {
					NOT = { is_in_same_market_area = $CEDER$.capital }
				}
			}
		}
	}
}
