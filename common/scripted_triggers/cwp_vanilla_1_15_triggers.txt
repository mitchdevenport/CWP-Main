# from ip4_negotiation_triggers.txt, on CWP's coup system (cwp_coup_events.1): an interest group could launch a
# coup instead of negotiating when one of its commanders is disloyal, strong enough and has a reason to
ig_could_coup_government = {
	owner = {
		NOT = { has_variable = cwp_coup_cooldown_variable }
	}
	any_scope_character = {
		OR = {
			has_role_of_type = general
			has_role_of_type = admiral
		}
		loyalty < define:NCharacters|LOYALTY_UNCERTAIN_THRESHOLD
		has_variable = monthly_cached_commander_coup_strength
		var:monthly_cached_commander_coup_strength >= {
			value = owner.cwp_country_coup_resistance
			multiply = 0.9
		}
		cwp_character_has_valid_reason_for_coup = yes
	}
}

# from 00_ai_triggers.txt
ai_ceding_state_would_isolate_states = {
	$STATE$ = { has_port_state = yes }
	$RECEIVER$ = {
		NOT = { market = $CEDER$.market }
	}
	NOT = {
		$CEDER$.market = {
			any_scope_country = {
				any_scope_state = {
					NOT = { this = $STATE$ }
					has_port_state = yes
					is_in_same_market_area = $STATE$
				}
			}
		}
	}
	OR = {
		AND = {
			$STATE$ = {
				NOT = { is_in_same_market_area = $CEDER$.capital }
				any_neighbouring_state = {
					is_in_same_market_area = $STATE$
				}
			}
		}
		AND = {
			$STATE$ = { is_in_same_market_area = $CEDER$.capital }
			$CEDER$ = {
				any_scope_state = {
					NOT = { is_in_same_market_area = $CEDER$.capital }
				}
			}
		}
	}
}
