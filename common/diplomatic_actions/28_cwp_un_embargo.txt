un_embargo = {
	groups = { general }

	requires_approval = no
	should_notify_third_parties = yes

	# Only placed by script on behalf of the United Nations
	selectable = {
		always = no
	}

	pact = {
		cost = 100

		is_hostile = yes
		is_embargo = yes
		is_two_sided_pact = no

		relations_progress_per_day = -1 # At -100, -1 relations

		requirement_to_maintain = {
			trigger = {
				has_diplomatic_relevance = scope:target_country
			}
		}

		requirement_to_maintain = {
			trigger = {
				scope:target_country = {
					NOT = { is_country_type = decentralized }
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				is_junior_in_customs_union = no
			}
		}

		requirement_to_maintain = {
			trigger = {
				scope:target_country = {
					infamy > 5
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				NOT = { scope:target_country = { has_law = law_type:law_isolationism } }
			}
		}

		requirement_to_maintain = {
			trigger = {
				custom_tooltip = {
					text = DIPLO_ACTION_RELATIONS_POOR_OR_LESS_NOT
					scope:target_country.relations:root <= relations_threshold:poor
				}
			}
		}

		manual_break_effect = {
			every_country = {
				limit = {
					has_variable = un_member
					has_diplomatic_pact = {
						who = scope:target_country
						type = un_embargo
					}
				}
				remove_diplomatic_pact = {
					country = scope:target_country
					type = un_embargo
				}
			}
		}

		auto_break_effect = {
			every_country = {
				limit = {
					has_variable = un_member
					has_diplomatic_pact = {
						who = scope:target_country
						type = un_embargo
					}
				}
				remove_diplomatic_pact = {
					country = scope:target_country
					type = un_embargo
				}
			}
		}
	}
}
