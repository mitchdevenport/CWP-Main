# CWP 1.5's Administer pact (removed with the 2025 restructure, restored for 1.14): a suzerain lends a subject bureaucracy,
# one point per 100,000 inhabitants of the subject, paid for with the same amount of its own bureaucracy.
administer = {
	groups = {
		overlord
	}
	requires_approval = no
	should_notify_third_parties = no
	unlocking_technologies = {
		international_relations_theory
	}

	potential = {
		NOT = {
			is_country_type = decentralized
		}
		scope:target_country = {
			NOT = {
				is_country_type = decentralized
			}
			is_subject_of = root
		}
	}

	possible = {
		NOT = {
			has_war_with = scope:target_country
		}
		has_diplomatic_relevance = scope:target_country
		is_revolutionary = no
		in_default = no
	}

	accept_effect = {
		add_overlord_lobby_appeasement_from_diplomacy = {
			FIRST = scope:target_country
			SECOND = root
			PRO_AMOUNT = 4
			ANTI_AMOUNT = -4
			FACTOR = appeasement_support_regime_started
		}
	}

	pact = {
		cost = 10
		is_two_sided_pact = no
		show_in_outliner = yes
		relations_progress_per_day = 1		# At 100, +1 relations
		relations_improvement_max = 80

		weekly_effect = {
			custom_tooltip = {
				text = administer_weekly_effect_tooltip
				scope:target_country = {
					remove_modifier = colonial_bureaucrats
					add_modifier = {
						name = colonial_bureaucrats
						days = 7
						multiplier = cwp_colonial_administration_amount
					}
				}
				# The suzerain pays for every subject it administers
				remove_modifier = colonial_administration
				add_modifier = {
					name = colonial_administration
					days = 7
					multiplier = cwp_colonial_administration_total
				}
			}
		}

		requirement_to_maintain = {
			trigger = {
				NOR = {
					has_war_with = scope:target_country
					is_country_type = decentralized
					in_default = yes
				}
				scope:target_country = {
					NOT = {
						is_country_type = decentralized
					}
					is_subject_of = root
				}
			}
		}

		manual_break_effect = {
			scope:target_country = {
				add_liberty_desire = 10
			}
			add_overlord_lobby_appeasement_from_diplomacy = {
				FIRST = scope:target_country
				SECOND = root
				PRO_AMOUNT = -4
				ANTI_AMOUNT = 4
				FACTOR = appeasement_support_regime_ended
			}
		}
	}

	ai = {
		max_influence_spending_fraction = 0.2
		propose_score = {
			value = 10
		}
		will_propose = {
			scaled_debt <= 0
			is_at_war = no
			has_attitude = {
				who = scope:target_country
				attitude = protective
			}
		}
		will_break = {
			OR = {
				scaled_debt > 0.25
				AND = {
					scaled_debt > 0
					is_at_war = yes
				}
				NOT = {
					has_attitude = {
						who = scope:target_country
						attitude = protective
					}
				}
			}
		}
	}
}
