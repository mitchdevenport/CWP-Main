# on_war_end (root = diplomatic play, scope:actor = initiator, scope:target = target)
# Costly wars force colonial empires to confront the future of their empire
cwp_un_war_end = {
	effect = {
		scope:actor ?= {
			effect_decolonization_after_war = yes
		}
		scope:target ?= {
			effect_decolonization_after_war = yes
		}
	}
}

# on_revolution_end (root = country that faced the revolution)
cwp_un_revolution_end = {
	effect = {
		set_variable = original_government
	}
}

# on_secession_end (root = country that faced the secession)
cwp_un_secession_end = {
	effect = {

	}
}

# on_civil_war_won (root = country)
cwp_un_civil_war_won = {
	events = {
		decolonization.666
	}
	effect = {
		# A victorious revolution lets go of the old regime's subjects
		if = {
			limit = {
				NOT = {
					has_variable = original_government
				}
			}
			add_modifier = {
				name = recent_revolution
				months = 12
				is_decaying = yes
			}
			effect_free_all_subjects = yes
		}
	}
}
