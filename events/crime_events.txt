# CWP: the base game's general flavour events, restored because CWP 1.5 had them
# (only the events CWP 1.5 used). Adapted to CWP's buildings, laws and the 1946
# technology level by docs/maintenance/restore_flavour_events.py; rerun it after game updates.
namespace = crime_events

crime_events.1 = { #Train Robbery
	type = country_event
	placement = scope:train_robbery_state

	title = crime_events.1.t
	desc = crime_events.1.d
	flavor = crime_events.1.f

	event_image = {
		video = "unspecific_trains"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	cooldown = { days = stupidly_long_modifier_time }

	trigger = {
		NOT = { has_variable = police_cooldown }
		institution_investment_level = {
			institution = institution_police
			value < 4
		}
		any_scope_state = {
			radical_fraction = {
				value > 0.40
				pop_type = laborers
			}
			has_building = building_railway
			NOT = {
				has_variable = train_robbery_variable
			}
			is_incorporated = no
			any_neighbouring_state = {
				owner = ROOT
				is_incorporated = yes
			}
		}
	}
	immediate = {
		set_variable = {
			name = police_cooldown
			years = 2
		}
		random_scope_state = {
			limit = {
				radical_fraction = {
					value > 0.40
					pop_type = laborers
				}
				has_building = building_railway
				NOT = {
					has_variable = train_robbery_variable
				}
				is_incorporated = no
				any_neighbouring_state = {
					owner = ROOT
					is_incorporated = yes
				}
			}
			save_scope_as = train_robbery_state
			set_variable = {
				name = train_robbery_variable
			}
		}
		generate_1_common_name = {
			CULTURE_1 = primary_culture
		}
	}
	option = {
		name = crime_events.1.a
		add_modifier = {
			name = funding_marshals
			days = normal_modifier_time
		}
	}
	option = {
		name = crime_events.1.b
		default_option = yes
		scope:train_robbery_state = {
			add_modifier = {
				name = modifier_train_robberies
				days = normal_modifier_time
			}
		}
	}
	after = {
		clear_generated_names = yes
	}
}

crime_events.2 = { #Organised crime found in state
	type = country_event
	placement = scope:organised_crime_state

	title = crime_events.2.t
	desc = crime_events.2.d
	flavor = crime_events.2.f

	event_image = {
		video = "unspecific_vandalized_storefront"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	icon = "gfx/interface/icons/event_icons/event_newspaper.dds"

	duration = 3

	cooldown = { days = stupidly_long_modifier_time }

	trigger = {
		NOT = { has_variable = police_cooldown }
		institution_investment_level = {
			institution = institution_police
			value < 4
		}
		any_scope_state = {
			radical_fraction = {
				value > 0.40
				pop_type = laborers
			}
			has_building = building_urban_center
			is_incorporated = yes
			NOR = {
				has_modifier = modifier_organised_crime_suppressed
				has_modifier = modifier_organised_crime
			}
		}
	}

	immediate = {
		set_variable = {
			name = police_cooldown
			years = 2
		}
		random_scope_state = {
			limit = {
				radical_fraction = {
					value > 0.40
					pop_type = laborers
				}
				is_incorporated = yes
				has_building = building_urban_center
				NOR = {
					has_modifier = modifier_organised_crime_suppressed
					has_modifier = modifier_organised_crime
				}
			}
			save_scope_as = organised_crime_state
		}
		set_variable = temperance_organised_crime_var
	}

	option = {
		name = crime_events.2.a
		add_modifier = {
			name = modifier_organised_crime_suppressed
			multiplier = money_amount_multiplier_small
			days = normal_modifier_time
		}
	}

	option = {
		name = crime_events.2.b
		default_option = yes
		add_modifier = {
			name = modifier_organised_crime
			days = normal_modifier_time
		}
	}
}
