# Corruption setup (master's debug.100 - debug.103)
# Fired one day after game start from history; corruption is only simulated for players
corruption_events.100 = {
	type = country_event
	hidden = yes

	trigger = {
		is_player = yes
	}

	immediate = {
		every_interest_group = {
			calc_ig_corruption = yes
		}
		if = {
			limit = {
				has_modifier = corruption_modifier_effects
			}
			remove_modifier = corruption_modifier_effects
		}
		add_modifier = {
			name = corruption_modifier_effects
			multiplier = corruption_modifier
		}
	}
}
