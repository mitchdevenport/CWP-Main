cwp_generic_economy_events.1 = {
	type = country_event
	title = cwp_generic_economy_events.1.t
	desc = cwp_generic_economy_events.1.desc
	flavor = cwp_generic_economy_events.1.f
	trigger = {
	}
	event_image = {
		video = "europenorthamerica_london_center"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	duration = 3
	immediate = {
		every_scope_state = {
			limit = {
				any_scope_building = {
					is_building_type = building_urban_center
					level >= 10
				}
			}
			set_variable = max_housing_state
		}
	}
	option = {
		# people love you
		name = cwp_generic_economy_events.1.a
		default_option = yes
		trigger = {
			has_variable = market_goods_more_expensive
			market_capital.market = {
				mg:services = {
					market_goods_cheaper > 0
				}
			}
		}
		hidden_effect = {
			every_scope_state = {
				limit = {
					has_variable = max_housing_state
				}
				add_loyalists_in_state = {
					value = 0.1
				}
				add_modifier = {
					name = more_housing
					days = very_long_modifier_time
				}
			}
		}
		custom_tooltip = max_housing_state_a_tt
	}
	option = {
		# small change since it was cheap anyway
		name = cwp_generic_economy_events.1.b
		trigger = {
			has_variable = market_goods_cheaper
			market_capital.market = {
				mg:services = {
					market_goods_cheaper > 0
				}
			}
		}
		hidden_effect = {
			every_scope_state = {
				limit = {
					has_variable = max_housing_state
				}
				add_loyalists_in_state = {
					value = 0.01
				}
			}
		}
		custom_tooltip = max_housing_state_b_tt
	}
	option = {
		# you fucked up lmao
		name = cwp_generic_economy_events.1.c
		trigger = {
			has_variable = market_goods_cheaper
			market_capital.market = {
				mg:services = {
					market_goods_cheaper < 0
				}
			}
		}
		hidden_effect = {
			every_scope_state = {
				limit = {
					is_incorporated = yes
				}
				add_radicals_in_state = {
					value = 0.1
				}
			}
		}
		custom_tooltip = max_housing_state_c_tt
	}
	option = {

cwp_generic_economy_events.2 = {
	type = country_event
	title = cwp_generic_economy_events.2.t
	desc = cwp_generic_economy_events.2.d
	flavor = cwp_generic_economy_events.2.f
	trigger = {
	}
	event_image = {
		video = "unspecific_trains"
	}
	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"
	icon = "gfx/interface/icons/event_icons/event_industry.dds"
	duration = 3
	immediate = {
		every_scope_state = {
			limit = {
				any_scope_building = {
					is_building_type = building_railway
					has_active_production_method = pm_high_speed_rail
					occupancy >= 0.75
					level >= 5
				}
			}
			set_variable = hsr_state
		}
	}
	option = {
		# hsr throughput
		name = cwp_generic_economy_events.2.a
		default_option = yes
		hidden_effect = {
			every_scope_state = {
				limit = {
					has_variable = hsr_state
				}
				add_modifier = {
					name = hsr_railway
					days = normal_modifier_time
				}
			}
		}
		custom_tooltip = hsr_state_tt
	}
	option = {
		# hsr-compatible industries throughput
		name = cwp_generic_economy_events.2.b
		hidden_effect = {
			every_scope_state = {
				limit = {
					has_variable = hsr_state
					any_scope_building = {
						OR = {
							is_building_type = building_coffee_plantation
							is_building_type = building_cotton_plantation
							is_building_type = building_tea_plantation
							is_building_type = building_tobacco_plantation
							is_building_type = building_sugar_plantation
							is_building_type = building_banana_plantation
							is_building_type = building_coal_mine
							is_building_type = building_oil_rig
							is_building_type = building_iron_mine
							is_building_type = building_bauxite_mine
							is_building_type = building_uranium_mine
							is_building_type = building_lead_mine
							is_building_type = building_sulfur_mine
							is_building_type = building_gold_mine
							is_building_type = building_logging_camp
						}
						OR = {
							has_active_production_method = pm_intermodal_containers_plantation
							has_active_production_method = pm_cold_chain_logistics_plantation
							has_active_production_method = pm_intermodal_containers_mines
							has_active_production_method = pm_double_stack_containers_mines
							has_active_production_method = pm_intermodal_containers_misc
						}
						occupancy >= 0.75
					}
				}
				add_modifier = {
					name = hsr_pm_state
					days = normal_modifier_time
				}
			}
		}
		custom_tooltip = hsr_pm_state_tt
	}
	option = {
		# mmmmm air research
		name = cwp_generic_economy_events.2.c
		trigger = {
			OR = {
				AND = {
					NOT = {
						has_technology_researched = jetliners
					}
					can_research = jetliners
				}
				AND = {
					NOT = {
						has_technology_researched = wide_body_jetliners
					}
					can_research = wide_body_jetliners
				}
			}
		}
		if = {
			limit = {
				NOT = {
					has_technology_researched = jetliners
				}
				can_research = jetliners
			}
			add_technology_progress = {
				progress = 4500
				technology = jetliners
			}
		}
		else_if = {
			limit = {
				NOT = {
					has_technology_researched = wide_body_jetliners
				}
				can_research = wide_body_jetliners
			}
			add_technology_progress = {
				progress = 5500
				technology = wide_body_jetliners
			}
		}
	}
}
