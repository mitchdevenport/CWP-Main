# CWP: the base game's general flavour events, restored because CWP 1.5 had them
# (only the events CWP 1.5 used). Adapted to CWP's buildings, laws and the 1946

namespace = communism

# Rising socialist leader
communism.7 = {
	type = country_event
	placement = root
	title = communism.7.t
	desc = communism.7.d
	flavor = communism.7.f

	duration = 3

	cooldown = { days = very_long_modifier_time }

	icon = "gfx/interface/icons/event_icons/event_portrait.dds"

	event_image = {
		video = "votp_agitator_speaking"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		# triggered by journal entry - je_communism_1 [on_monthly_pulse]
		# triggered by on_action - on_yearly_events
		NOT = { has_law_or_variant = law_type:law_council_republic }
		any_interest_group = {
			ig_counts_as_marginal = no
			any_scope_character = {
				OR = {
					has_role_of_type = politician
					has_role_of_type = agitator
				}
				NOT = { has_modifier = popular_socialist_leader }
				OR = {
					has_ideology = ideology:ideology_communist
					has_ideology = ideology:ideology_anarchist
					has_ideology = ideology:ideology_vanguardist
				}
			}
		}
	}

	immediate = {
		random_interest_group = {
			limit = {
				ig_counts_as_marginal = no
				any_scope_character = {
					OR = {
						has_role_of_type = politician
						has_role_of_type = agitator
					}
					NOT = { has_modifier = popular_socialist_leader }
					OR = {
						has_ideology = ideology:ideology_communist
						has_ideology = ideology:ideology_anarchist
						has_ideology = ideology:ideology_vanguardist
					}
				}
			}
			save_scope_as = popular_socialist_ig
			random_scope_character = {
				limit = {
					OR = {
						has_role_of_type = politician
						has_role_of_type = agitator
					}
					NOT = { has_modifier = popular_socialist_leader }
					OR = {
						has_ideology = ideology:ideology_communist
						has_ideology = ideology:ideology_anarchist
						has_ideology = ideology:ideology_vanguardist
					}
				}
				save_scope_as = popular_socialist_leader
				set_temporary_event_immortality = { DURATION = 3 }
			}
		}
	}

	after = {
		revoke_temporary_event_immortality = { CHARACTER = scope:popular_socialist_leader }
	}

	option = {
		name = communism.7.a
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ai_avoids_radicals = yes
				}
				add = 10
			}
		}
		trigger = {
			scope:popular_socialist_ig = {
				is_in_government = yes
			}
		}
		default_option = yes
		scope:popular_socialist_leader = {
			add_modifier = {
				name = popular_socialist_leader
				days = long_modifier_time
			}
		}
	}

	option = {
		name = communism.7.b
		ai_chance = {
			base = 1
			modifier = {
				trigger = {
					ai_wants_radicals = yes
				}
				add = 10
			}
		}
		trigger = {
			scope:popular_socialist_ig = {
				is_in_government = no
			}
		}
		default_option = yes
		scope:popular_socialist_leader = {
			add_modifier = {
				name = popular_socialist_leader
				days = long_modifier_time
			}
		}
		random_scope_state = {
			limit = {
				is_incorporated = yes
			}
			add_radicals_in_state = {
				value = large_radicals
				strata = lower
			}
		}
	}
}

# The Red Army
communism.8 = {
	type = country_event
	placement = root
	title = communism.8.t
	desc = communism.8.d
	flavor = communism.8.f

	duration = 3

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	fire_only_once = yes

	trigger = {
		# triggered by on_action - on_monthly_pulse_country
		has_law_or_variant = law_type:law_council_republic
		always = no
		any_scope_character = {
			OR = {
				has_variable = red_army_candidate
				AND = {
					OR = {
						is_interest_group_type = ig_trade_unions
						has_socialist_ideology = yes
					}
					OR = {
						has_role_of_type = general
						has_role_of_type = admiral
					}
				}
			}
		}
	}

	immediate = {
		ordered_scope_character = {
			limit = {
				OR = {
					has_variable = red_army_candidate
					AND = {
						OR = {
							is_interest_group_type = ig_trade_unions
							has_socialist_ideology = yes
						}
						OR = {
							has_role_of_type = general
							has_role_of_type = admiral
						}
					}
				}
			}
			order_by = prominence
			position = 0
			save_scope_as = red_army_general
			set_temporary_event_immortality = { DURATION = 3 }
		}
		ig:ig_armed_forces ?= {
			save_scope_as = armed_forces_ig
		}
	}

	option = {
		name = communism.8.a
		ig:ig_armed_forces ?= {
			custom_tooltip = red_army_tt
			hidden_effect = {
				set_interest_group_name = ig_red_army
				if = {
					limit = {
						always = no
					}
				}
				if = {
					limit = {
						has_ideology = ideology:ideology_jingoist
					}
					remove_ideology = ideology_jingoist
				}
				if = {
					limit = {
						has_ideology = ideology:ideology_patriotic
					}
					remove_ideology = ideology_patriotic
					add_ideology = ideology_proletarian
				}
				if = {
					limit = {
						has_ideology = ideology:ideology_republican
					}
					remove_ideology = ideology_republican
					add_ideology = ideology_proletarian
				}
				custom_tooltip = {
					text = loyalist_change_stances_tt
					if = {
						limit = {
							has_ideology = ideology:ideology_loyalist
						}
						remove_ideology = ideology_loyalist
						add_ideology = ideology_proletarian
					}
				}
				add_ideology = ideology_proletarian
				owner = {
					set_variable = ig_red_army_active
				}
			}
		}
		if = {
			limit = {
				NOT = {
					scope:red_army_general ?= {
						is_interest_group_type = ig_armed_forces
					}
				}
			}
			scope:red_army_general ?= {
				set_interest_group = scope:armed_forces_ig
				custom_tooltip = {
					text = leader_of_red_army_tt
					set_as_interest_group_leader = yes
				}
			}
		}
		every_scope_character = {
			limit = {
				OR = {
					has_variable = red_army_candidate
					AND = {
						OR = {
							is_interest_group_type = ig_trade_unions
							has_socialist_ideology = yes
						}
						OR = {
							has_role_of_type = general
							has_role_of_type = admiral
						}
					}
				}
				NOT = {
					scope:red_army_general ?= this
				}
			}
			if = {
				limit = {
					has_variable = red_army_candidate
				}
				set_interest_group = scope:armed_forces_ig
			}
			else = {
				random = {
					chance = {
						value = 50
					}
					set_interest_group = scope:armed_forces_ig
				}
			}
		}
	}

	after = {
		hidden_effect = {
			scope:red_army_general ?= {
				set_as_interest_group_leader = yes
			}
		}
		revoke_temporary_event_immortality = { CHARACTER = scope:red_army_general }
	}
}
