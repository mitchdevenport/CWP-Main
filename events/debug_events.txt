namespace = debug

# Chaos mode game rule (cwp_chaos_mode): on the first monthly pulse the player takes over the whole world,
# which then falls apart through decolonization.
debug.999 = {
	hidden = yes
	type = country_event
	trigger = {
		is_player = yes
		has_game_rule = cwp_chaos_mode_on
		NOT = { has_global_variable = cwp_chaos_mode }
	}
	immediate = {
		set_global_variable = cwp_chaos_mode
		if = {
			limit = { is_subject = yes }
			make_independent = yes
		}
		every_state = {
			limit = { is_capital = no }
			set_state_type = unincorporated
		}
		every_country = {
			limit = { NOT = { THIS = ROOT } }
			ROOT = { annex = PREV }
		}
		effect_decolonize_all = yes
		every_subject_or_below = {
			make_independent = yes
		}
	}
}
