namespace = communism

# CWP: the base game's Red Army event as CWP 1.5 had it (a council republic whose Armed Forces are led by a communist
# or vanguardist renames them the Red Army). Victoria 3 1.15 rebuilt the base game's event around its Revolutionary
# Guard law and new ideologies, which CWP does not have, so docs/maintenance/restore_flavour_events.py leaves it out
# of events/communism.txt (PINNED) and keeps its monthly hook. The base game's texts are used; they name the
# armed_forces_ig scope.

# The Red Army
communism.8 = {
	type = country_event
	placement = root
	title = communism.8.t
	desc = communism.8.d
	flavor = communism.8.f

	duration = 3

	icon = "gfx/interface/icons/event_icons/event_military.dds"

	event_image = {
		video = "unspecific_military_parade"
	}

	on_created_soundeffect = "event:/SFX/UI/Alerts/event_appear"

	trigger = {
		has_law_or_variant = law_type:law_council_republic
		NOT = {	has_variable = red_army_var }
		ig:ig_armed_forces ?= {
			leader = {
				OR = {
					has_ideology = ideology:ideology_communist
					has_ideology = ideology:ideology_vanguardist
				}
			}
		}
	}

	immediate = {
		ig:ig_armed_forces ?= {
			save_scope_as = red_army_ig
			save_scope_as = armed_forces_ig
			leader = {
				save_scope_as = red_army_general
			}
		}
		set_variable = red_army_var
	}

	option = {
		name = communism.8.a
		ig:ig_armed_forces ?= {
			custom_tooltip = cwp_red_army_tt
			hidden_effect = {
				set_interest_group_name = ig_red_army
			}
			remove_ideology = ideology_loyalist
			add_ideology = ideology_proletarian
		}
	}
}
