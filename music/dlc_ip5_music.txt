# CWP version of the base game's DLC music list: Victoria 3 1.15 weighs these tracks by base-game geographic regions,
# which CWP's map does not have, so they are expressed with CWP's (Russia and Siberia; a capital in the Japanese homeland)
##### MOOD TRACKS

mx_mood_tr_71 = {
	music = "event:/MUSIC/DLC/IP5/Mood/01_Notre_Pays_MOODVERSION"
	name = "01_Notre_Pays_MOODVERSION"
	pause_factor = 50
	mood = yes	
	can_be_interrupted = yes
	trigger_prio_override = no
	is_valid = { has_ip5_soundtrack_dlc_trigger = yes }
	weight = {
		value = 100
		if = {
			limit = {
				any_primary_culture = { has_discrimination_trait = heritage_east_slavic }
			}
			add = 100
		}
		if = {
			limit = {
				OR = { is_in_geographic_region = geographic_region_russia is_in_geographic_region = geographic_region_siberia }
				NOT = { capital ?= { state_region = { is_homeland = cu:japanese } } }
			}
			add = 100
		}
		if = {
			limit = {
				NOT = { OR = { is_in_geographic_region = geographic_region_russia is_in_geographic_region = geographic_region_siberia } }
				overlord ?= {
					any_primary_culture = { has_discrimination_trait = heritage_east_slavic }
				}
			}
			add = 50
		}
	}
}
